// 모바일 웹 이미지 저장 / 공유 안전 다운로드 로직 보완 async function shareOrSaveImage(mode) { try { await generateHappinessImage(mode, async (canvas, blob) => { const fileName = mode === 'wallpaper' ? `${userName}_행복_10계명_배경화면.png` : `${userName}_행복_10계명_카드.png`; const file = new File([blob], fileName, { type: 'image/png' }); // 1. 모바일 브라우저 공유 지원 확인 if (navigator.canShare && navigator.canShare({ files: [file] })) { try { await navigator.share({ files: [file], title: `${userName}의 행복 10계명`, text: `내가 직접 작성한 ${userName}의 행복 10계명입니다.` }); return; } catch (err) { // 취소(AbortError) 이외의 경우에만 파일 직접 다운로드 시도 if (err.name !== 'AbortError') { console.log('공유 실패, 다운로드로 진행합니다.'); } else { return; } } } // 2. Web Share API 미지원 및 실패 시 직접 다운로드 const link = document.createElement('a'); link.download = fileName; link.href = URL.createObjectURL(blob); document.body.appendChild(link); link.click(); document.body.removeChild(link); setTimeout(() => URL.revokeObjectURL(link.href), 1000); }); } catch (err) { console.error('이미지 생성 중 오류가 발생했습니다:', err); alert('이미지 생성 중 오류가 발생했습니다. 다시 시도해주세요.'); } } // 새로운 이미지 저장하기 함수 (카드 이미지만 다운로드) async function downloadImage() { try { await generateHappinessImage('card', async (canvas, blob) => { const fileName = `${userName}_행복_10계명_카드.png`; const link = document.createElement('a'); link.download = fileName; link.href = URL.createObjectURL(blob); document.body.appendChild(link); link.click(); document.body.removeChild(link); setTimeout(() => URL.revokeObjectURL(link.href), 1000); }); } catch (err) { console.error('이미지 저장 중 오류가 발생했습니다:', err); alert('이미지 저장 중 오류가 발생했습니다. 다시 시도해주세요.'); } }
